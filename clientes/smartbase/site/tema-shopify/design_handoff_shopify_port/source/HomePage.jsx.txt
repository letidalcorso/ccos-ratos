// Optional highlight band (launches / promos) — toggled 0–2 from Tweaks. Lives
// INSIDE the showcase stage (before the products), on the photo. Dark liquid
// glass. Carries the product info itself (no product page) + first-person CTA.
const HL_IMG = ["assets/products/iphone-17-pro.png", "assets/products/airpods-pro.png"];
const HL_DEFS = [
  { kick: "lançamento", name: "iPhone 17 Pro", sub: "o topo da linha, lacrado e sob encomenda — chega em até 7 dias.",
    info: "256GB · 512GB · 1TB ; cosmic orange ; garantia apple · 12 meses" },
  { kick: "promo da semana", name: "AirPods Pro 3", sub: "cancelamento de ruído com condição especial por tempo limitado.",
    info: "usb-c · estojo magsafe ; seminovo conferido ; garantia smartbase · 3 meses" },
];

function HighlightsBand({ go, tw }) {
  const w = tw || {};
  const n = Math.max(0, Math.min(2, Number(w.hlCount) || 0));
  if (!n) return null;
  const blocks = [1, 2].slice(0, n).map((i) => {
    const d = HL_DEFS[i - 1];
    const dest = w["hl" + i + "Dest"] || "vitrine";
    const cta = dest === "whatsapp" ? "quero esse" : dest === "link próprio" ? "quero saber mais" : "quero ver na vitrine";
    const link = dest === "whatsapp"
      ? { href: waHref(), target: "_blank", rel: "noopener noreferrer" }
      : dest === "link próprio"
        ? { href: (w["hl" + i + "Url"] || "").trim() || "#", target: "_blank", rel: "noopener noreferrer" }
        : { onClick: () => go("vitrine") };
    return {
      id: "hl" + i,
      kick: (w["hl" + i + "Title"] || "").trim() || d.kick,
      name: (w["hl" + i + "Name"] || "").trim() || d.name,
      sub: (w["hl" + i + "Sub"] || "").trim() || d.sub,
      info: ((w["hl" + i + "Info"] || "").trim() || d.info).split(";").map((s) => s.trim()).filter(Boolean),
      img: HL_IMG[i - 1], cta, link,
    };
  });
  return (
    <div className={"hms-hl-band-wrap" + (n === 2 ? " two" : "")}>
      {blocks.map((b) => (
        <a key={b.id} className="hms-hl-band" {...b.link}>
          <div className="hms-hl-band-media">
            <image-slot id={"home-" + b.id} class="hms-hl-band-img" src={b.img} fit="contain" placeholder="foto do destaque"></image-slot>
          </div>
          <div className="hms-hl-band-copy">
            <span className="hms-hl-band-kick">{b.kick}</span>
            <h3 className="hms-hl-band-t">{b.name}</h3>
            <p className="hms-hl-band-d">{b.sub}</p>
            <div className="hms-hl-band-meta">{b.info.map((s, k) => <span key={k}>{s}</span>)}</div>
            <span className="hms-hl-band-cta">{b.cta} <span aria-hidden="true">→</span></span>
          </div>
        </a>
      ))}
    </div>
  );
}
