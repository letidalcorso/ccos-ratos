function FaqRow({ q, a, open, onToggle }) {
  return (
    <div className={"faq-row rise" + (open ? " open" : "")}>
      <button className="faq-q" onClick={onToggle}>
        <span>{q}</span><span className="faq-sign">{open ? "–" : "+"}</span>
      </button>
      <div className="faq-a" style={{ maxHeight: open ? 260 : 0 }}><p>{a}</p></div>
    </div>
  );
}

function FaqPage({ go }) {
  const [open, setOpen] = React.useState("0-0");
  return (

      <section className="page faq-body">
        {FAQ.map((group, gi) => (
          <div key={gi} className="faq-group rise">
            <h3 className="faq-cat">{group.cat}</h3>
            {group.compare && <GarantiaCompare />}
            {group.items.map((it, ii) => {
