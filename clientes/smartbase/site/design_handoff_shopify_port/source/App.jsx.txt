const TWEAK_DEFAULTS = /*EDITMODE-BEGIN*/{
  "glass": 100,
  "radius": 100,
  "accent": "#3E82F7",
  "glow": true,
  "sbScale": 100,
  "sbX": 0,
  "sbY": 0,
  "sbOpacity": 85,
  "sbShine": 100,
  "liquidAnim": true,
  "hlCount": 0,
  "hl1Title": "",
  "hl1Name": "",
  "hl1Sub": "",
  "hl1Info": "",
  "hl1Dest": "vitrine",
  "hl1Url": "",
  "hl2Title": "",
  "hl2Name": "",
  "hl2Sub": "",
  "hl2Info": "",
  "hl2Dest": "whatsapp",
  "hl2Url": "",
  "motion": "equilibrado"
}/*EDITMODE-END*/;

function App() {
  const [t, setTweak] = useTweaks(TWEAK_DEFAULTS);
  const [page, setPage] = React.useState(() => location.hash.replace("#", "") || "home");
  const go = (p) => { setPage(p); location.hash = p; };

  // every route change starts at the top — run after the new page paints,
  // before the browser shows it (no smooth animation across a layout swap).
  React.useLayoutEffect(() => {
    window.scrollTo(0, 0);
  }, [page]);

  // apply tweaks as CSS variables on :root
  React.useEffect(() => {
    const r = document.documentElement.style;
    const g = t.glass / 100;
    r.setProperty("--sb-lg-blur", `blur(${Math.round(18 + g * 30)}px) saturate(${120 + g * 60}%)`);
    r.setProperty("--sb-lg-rim", `rgba(255,255,255,${(0.30 + g * 0.30).toFixed(2)})`);
    const rs = t.radius / 100;
    r.setProperty("--sb-r-card", Math.round(28 * rs) + "px");
    r.setProperty("--sb-r-panel", Math.round(22 * rs) + "px");
    r.setProperty("--sb-r-lg", Math.round(36 * rs) + "px");
    r.setProperty("--sb-r-sm", Math.round(16 * rs) + "px");
    r.setProperty("--sb-signal", t.accent);
    document.body.classList.toggle("no-glow", !t.glow);
    // sb watermark live controls
    r.setProperty("--sb-mark-scale", (t.sbScale / 100).toFixed(3));
    r.setProperty("--sb-mark-x", t.sbX + "px");
    r.setProperty("--sb-mark-y", t.sbY + "px");
    r.setProperty("--sb-mark-opacity", (t.sbOpacity / 100).toFixed(2));
    r.setProperty("--sb-mark-shine", (t.sbShine / 100).toFixed(2));
    // liquid-glass showcase: motion on/off
    document.body.classList.toggle("hms-no-anim", !t.liquidAnim);
    // movimento: intensidade dos reveals (CSS vars) + força do parallax
    const MO = { sutil: [30, 0.78, 0, 0.05], equilibrado: [66, 0.98, 0, 0.08], forte: [108, 1.18, 5, 0.13] }[t.motion] || [66, 0.98, 0, 0.08];
    r.setProperty("--rv-y", MO[0] + "px");
    r.setProperty("--rv-dur", MO[1] + "s");
    r.setProperty("--rv-blur", MO[2] + "px");
    window.__parK = MO[3];
  }, [t.glass, t.radius, t.accent, t.glow, t.sbScale, t.sbX, t.sbY, t.sbOpacity, t.sbShine, t.liquidAnim, t.motion]);

  React.useEffect(() => {
    const onHash = () => setPage(location.hash.replace("#", "") || "home");
    addEventListener("hashchange", onHash);
    return () => removeEventListener("hashchange", onHash);
  }, []);

  // reveal-on-scroll + nav-theme detection. Resilient: re-asserts after
  // post-mount re-renders (useTweaks/localStorage) that would otherwise reset
  // React-controlled className.
  React.useEffect(() => {
    let raf;
    // Toggle a light-on-nav theme when the continuous light zone passes behind
    // the fixed chrome bar / sidebar, so their glass + icons stay legible.
    const navTheme = () => {
      const zone = document.querySelector(".hm-light-zone");
      const b = document.body;
      if (!zone) { b.classList.remove("chrome-light", "sidebar-light"); return; }
      const r = zone.getBoundingClientRect();
      const over = (y) => r.top <= y && r.bottom >= y;
      b.classList.toggle("chrome-light", over(44));                 // chrome bar centre
      b.classList.toggle("sidebar-light", over(window.innerHeight / 2)); // sidebar centre
    };
    const prefersReduce = matchMedia("(prefers-reduced-motion: reduce)").matches;
    const reveal = () => {
      document.querySelectorAll(".rise:not(.in)").forEach((el) => {
        const r = el.getBoundingClientRect();
        if (r.top < window.innerHeight * 0.88 && r.bottom > 0) el.classList.add("in");
      });
      const pk = window.__parK || 0;
      if (pk && !prefersReduce) {
        document.querySelectorAll("[data-par]").forEach((el) => {
          const s = parseFloat(el.getAttribute("data-par")) || 1;
          const r = el.getBoundingClientRect();
          const d = (r.top + r.height / 2) - window.innerHeight / 2;
          el.style.transform = `translate3d(0, ${(-d * pk * s).toFixed(1)}px, 0)`;
        });
      }
      navTheme();
    };
    reveal();
    const onScroll = () => { cancelAnimationFrame(raf); raf = requestAnimationFrame(reveal); };
    addEventListener("scroll", onScroll, { passive: true });
    addEventListener("resize", onScroll, { passive: true });
    const timers = [60, 250, 600].map((ms) => setTimeout(reveal, ms));
    return () => {
      removeEventListener("scroll", onScroll);
      removeEventListener("resize", onScroll);
      timers.forEach(clearTimeout);
      cancelAnimationFrame(raf);
    };
  }, [page]);

  const urls = { home: "smartbasebr.com", vitrine: "smartbasebr.com/vitrine", faq: "smartbasebr.com/faq",
    contato: "smartbasebr.com/contato", blog: "smartbasebr.com/blog", post: "smartbasebr.com/blog/post" };

  return (
    <div className="site">
      <BrowserBar url={urls[page] || "smartbasebr.com"} />
      <Sidebar page={page} go={go} />
      <div className="page-shell" key={page}>
        {page === "home" && <HomePage go={go} tw={t} />}
        {page === "vitrine" && <VitrinePage go={go} tw={t} />}
        {page === "faq" && <FaqPage go={go} />}
        {page === "contato" && <ContatoPage go={go} />}
        {page === "blog" && <BlogPage go={go} />}
        {page === "post" && <PostPage go={go} />}
      </div>
      <CtaBand go={go} page={page} />
      <Footer go={go} />
      <TweaksPanel>
        <TweakSection label="Movimento" />
        <TweakRadio label="Intensidade" value={t.motion} options={["sutil", "equilibrado", "forte"]}
          onChange={(v) => setTweak("motion", v)} />
        <TweakSection label="Liquid Glass" />
        <TweakSlider label="Intensidade do vidro" value={t.glass} min={0} max={100} unit="%"
          onChange={(v) => setTweak("glass", v)} />
        <TweakSlider label="Raio dos cantos" value={t.radius} min={50} max={160} unit="%"
          onChange={(v) => setTweak("radius", v)} />
        <TweakToggle label="Brilho do fundo (iOS)" value={t.glow}
          onChange={(v) => setTweak("glow", v)} />
        <TweakSection label="Acento" />
        <TweakColor label="Azul signal" value={t.accent}
          options={["#3E82F7", "#0A84FF", "#00A8E8", "#5E5CE6"]}
          onChange={(v) => setTweak("accent", v)} />
        <TweakSection label="Marca “sb”" />
        <TweakSlider label="Tamanho" value={t.sbScale} min={40} max={170} unit="%"
          onChange={(v) => setTweak("sbScale", v)} />
        <TweakSlider label="Posição horizontal" value={t.sbX} min={-400} max={400} unit="px"
          onChange={(v) => setTweak("sbX", v)} />
        <TweakSlider label="Posição vertical" value={t.sbY} min={-500} max={500} unit="px"
          onChange={(v) => setTweak("sbY", v)} />
        <TweakSlider label="Opacidade" value={t.sbOpacity} min={5} max={100} unit="%"
          onChange={(v) => setTweak("sbOpacity", v)} />
        <TweakSlider label="Brilho do vidro" value={t.sbShine} min={0} max={200} unit="%"
          onChange={(v) => setTweak("sbShine", v)} />
        <TweakSection label="Vitrine (liquid glass)" />
        <TweakToggle label="Animação dos produtos" value={t.liquidAnim}
          onChange={(v) => setTweak("liquidAnim", v)} />
        <TweakRadio label="Blocos de destaque" value={String(t.hlCount)} options={["0", "1", "2"]}
          onChange={(v) => setTweak("hlCount", Number(v))} />
        {t.hlCount >= 1 && <>
          <TweakText label="Destaque 1 · etiqueta" value={t.hl1Title} placeholder="lançamento"
            onChange={(v) => setTweak("hl1Title", v)} />
          <TweakText label="Destaque 1 · produto" value={t.hl1Name} placeholder="iPhone 17 Pro"
            onChange={(v) => setTweak("hl1Name", v)} />
          <TweakText label="Destaque 1 · texto" value={t.hl1Sub} placeholder="o novo iphone chegou…"
            onChange={(v) => setTweak("hl1Sub", v)} />
          <TweakText label="Destaque 1 · specs (separe com ;)" value={t.hl1Info} placeholder="256GB ; cosmic orange ; garantia 12 meses"
            onChange={(v) => setTweak("hl1Info", v)} />
          <TweakSelect label="Destaque 1 · vai pra" value={t.hl1Dest} options={["vitrine", "whatsapp", "link próprio"]}
            onChange={(v) => setTweak("hl1Dest", v)} />
          {t.hl1Dest === "link próprio" &&
            <TweakText label="Destaque 1 · URL" value={t.hl1Url} placeholder="https://…"
              onChange={(v) => setTweak("hl1Url", v)} />}
        </>}
        {t.hlCount >= 2 && <>
          <TweakText label="Destaque 2 · etiqueta" value={t.hl2Title} placeholder="promo da semana"
            onChange={(v) => setTweak("hl2Title", v)} />
          <TweakText label="Destaque 2 · produto" value={t.hl2Name} placeholder="AirPods Pro 3"
            onChange={(v) => setTweak("hl2Name", v)} />
          <TweakText label="Destaque 2 · texto" value={t.hl2Sub} placeholder="condições especiais…"
            onChange={(v) => setTweak("hl2Sub", v)} />
          <TweakText label="Destaque 2 · specs (separe com ;)" value={t.hl2Info} placeholder="usb-c ; seminovo ; garantia 3 meses"
            onChange={(v) => setTweak("hl2Info", v)} />
          <TweakSelect label="Destaque 2 · vai pra" value={t.hl2Dest} options={["vitrine", "whatsapp", "link próprio"]}
            onChange={(v) => setTweak("hl2Dest", v)} />
          {t.hl2Dest === "link próprio" &&
            <TweakText label="Destaque 2 · URL" value={t.hl2Url} placeholder="https://…"
              onChange={(v) => setTweak("hl2Url", v)} />}
        </>}
      </TweaksPanel>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById("root")).render(<App />);
